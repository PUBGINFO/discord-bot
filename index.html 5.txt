<script>
  const API_BASE = "https://discord-bot.pubgmk14-kr.workers.dev";
  let selectedRoleIds = new Set();
  let currentRoles = [];

  // 안정적인 세션 토큰 관리 (새로고침 시 풀림 현상 원천 차단)
  function initToken() {
    const urlParams = new URLSearchParams(location.search);
    const urlSession = urlParams.get("session");
    if (urlSession) {
      localStorage.setItem("session", urlSession);
      window.history.replaceState({}, document.title, location.pathname);
      return urlSession;
    }
    return localStorage.getItem("session");
  }

  let token = initToken();

  const showModal = (title, message, onConfirm, showCancel = false) => {
    document.getElementById('modal-title').textContent = title;
    document.getElementById('modal-message').textContent = message;
    const cancelBtn = document.getElementById('modal-cancel');
    const confirmBtn = document.getElementById('modal-confirm');
    cancelBtn.style.display = showCancel ? 'inline-block' : 'none';

    const modal = document.getElementById('custom-modal');
    modal.classList.add('show');

    const newConfirm = confirmBtn.cloneNode(true);
    confirmBtn.parentNode.replaceChild(newConfirm, confirmBtn);
    const newCancel = cancelBtn.cloneNode(true);
    cancelBtn.parentNode.replaceChild(newCancel, cancelBtn);

    newConfirm.onclick = () => { modal.classList.remove('show'); if (onConfirm) onConfirm(); };
    newCancel.onclick = () => modal.classList.remove('show');
  };

  async function api(path, options = {}) {
    const headers = { ...(options.headers || {}), Authorization: `Bearer ${token}`, "Content-Type": "application/json" };
    const res = await fetch(`${API_BASE}${path}`, { ...options, headers });
    const data = await res.json().catch(() => ({}));
    if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`);
    return data;
  }

  async function init() {
    if (!token) {
      document.getElementById('user-status').textContent = "로그인하지 않았습니다.";
      return;
    }
    try {
      const me = await api("/api/me");
      const statusEl = document.getElementById('user-status');
      statusEl.innerHTML = `<span class="dot"></span>${escapeHtml(me.username)}`;
      statusEl.classList.add('online');

      const loginBtn = document.getElementById('login-btn');
      loginBtn.textContent = "로그아웃";
      loginBtn.className = "btn btn-danger";
      loginBtn.onclick = () => { 
        localStorage.removeItem("session"); 
        token = null;
        location.href = location.pathname; 
      };

      const guilds = await api("/api/guilds");
      const guildSelect = document.getElementById('guild-select');
      guildSelect.innerHTML = guilds.map(g => `<option value="${g.id}">${escapeHtml(g.name)}</option>`).join("");

      await loadChannels();
      await loadRoles();
      await loadSchedules();
    } catch (e) {
      // 서버에서 유효하지 않은 세션이라고 판단했을 때만 안전하게 로그아웃 처리
      document.getElementById('user-status').textContent = "세션이 만료되었습니다.";
      localStorage.removeItem("session");
      token = null;
    }
  }

  document.getElementById('login-btn').onclick = () => {
    if (token) return;
    location.href = `${API_BASE}/api/login`;
  };

  document.getElementById('guild-select').onchange = () => { loadChannels(); loadRoles(); };

  async function loadChannels() {
    const guildId = document.getElementById('guild-select').value;
    if (!guildId) return;
    try {
      const channels = await api(`/api/guilds/${guildId}/channels`);
      document.getElementById('channel-select').innerHTML = channels.map(c => `<option value="${c.id}"># ${escapeHtml(c.name)}</option>`).join("");
    } catch (e) {
      document.getElementById('channel-select').innerHTML = `<option value="">채널 로드 실패 (봇 권한 확인)</option>`;
    }
  }

  async function loadRoles() {
    const guildId = document.getElementById('guild-select').value;
    const list = document.getElementById('role-chip-list');
    selectedRoleIds = new Set();
    if (!guildId) { list.innerHTML = '<span class="role-chip-empty">서버를 선택하면 역할 목록이 표시됩니다</span>'; return; }

    try {
      currentRoles = await api(`/api/guilds/${guildId}/roles`);
      if (!currentRoles.length) {
        list.innerHTML = '<span class="role-chip-empty">멘션 가능한 역할이 없습니다</span>';
        return;
      }
      list.innerHTML = currentRoles.map(r => `
        <button type="button" class="role-chip" data-role-id="${r.id}">
          <span class="swatch" style="${r.color ? `background:${r.color}` : ''}"></span>
          ${escapeHtml(r.name)}
        </button>
      `).join("");

      list.querySelectorAll('.role-chip').forEach(chip => {
        chip.onclick = () => {
          const id = chip.dataset.roleId;
          if (selectedRoleIds.has(id)) { selectedRoleIds.delete(id); chip.classList.remove('selected'); }
          else { selectedRoleIds.add(id); chip.classList.add('selected'); }
        };
      });
    } catch (e) {
      list.innerHTML = '<span class="role-chip-empty">역할 로드 실패 (봇 권한 확인)</span>';
    }
  }

  document.getElementById('message-input').oninput = (e) => {
    document.getElementById('char-count').textContent = `${e.target.value.length} / 2000`;
  };

  document.getElementById('create-btn').onclick = async () => {
    if (!token) { showModal('로그인 필요', '예약을 생성하려면 먼저 로그인이 필요합니다.'); return; }
    const statusMsg = document.getElementById('status-msg');
    try {
      statusMsg.textContent = "저장 중...";
      
      const rawDate = document.getElementById('start-at-input').value;
      if (!rawDate) throw new Error("시작 날짜와 시간을 선택해주세요.");
      
      const kstDate = new Date(rawDate + "+09:00");

      await api("/api/schedules", {
        method: "POST",
        body: JSON.stringify({
          guildId: document.getElementById('guild-select').value,
          channelId: document.getElementById('channel-select').value,
          message: document.getElementById('message-input').value,
          roleIds: Array.from(selectedRoleIds),
          startAt: kstDate.toISOString(),
          timezone: "Asia/Seoul",
          months: Number(document.getElementById('months-input').value || 0),
          days: Number(document.getElementById('days-input').value || 1),
          hours: Number(document.getElementById('hours-input').value || 0),
          minutes: Number(document.getElementById('minutes-input').value || 0)
        })
      });
      statusMsg.textContent = "예약이 생성되었습니다.";
      document.getElementById('message-input').value = "";
      document.getElementById('char-count').textContent = "0 / 2000";
      document.querySelectorAll('.role-chip.selected').forEach(c => c.classList.remove('selected'));
      selectedRoleIds = new Set();
      await loadSchedules();
    } catch (e) {
      statusMsg.textContent = e.message;
    }
  };

  function roleNamesFor(idString) {
    if (!idString) return [];
    const ids = idString.split(",");
    return ids.map(id => {
      const r = currentRoles.find(r => r.id === id);
      return r ? r.name : null;
    }).filter(Boolean);
  }

  function formatKstTime(utcIsoString) {
    try {
      const d = new Date(utcIsoString);
      return d.toLocaleString("ko-KR", {
        timeZone: "Asia/Seoul",
        year: "numeric",
        month: "long",
        day: "numeric",
        hour: "2-digit",
        minute: "2-digit",
        hour12: true
      });
    } catch (e) {
      return utcIsoString;
    }
  }

  async function loadSchedules() {
    const schedules = await api("/api/schedules");
    const container = document.getElementById('schedules-list');

    if (!schedules.length) {
      container.innerHTML = '<p class="empty-state">등록된 예약이 없습니다. 위에서 첫 예약을 만들어보세요.</p>';
      return;
    }

    container.innerHTML = `<div class="timeline">` + schedules.map(s => {
      const enabled = Number(s.enabled);
      const roleNames = roleNamesFor(s.role_id);
      const formattedTime = formatKstTime(s.next_run_at);
      return `
      <div class="tl-item ${enabled ? '' : 'paused'}">
        <div class="tl-node"></div>
        <div class="tl-card">
          <div class="tl-message">${escapeHtml(s.message)}</div>
          <div class="tl-meta">
            <span class="badge ${enabled ? 'on' : 'off'}"><span class="dot"></span>${enabled ? '활성화' : '일시중지'}</span>
            <span class="tl-time">${escapeHtml(formattedTime)}</span>
            <span class="sep">·</span>
            <span>${s.months}M ${s.days}D ${s.hours}H ${s.minutes}Min 반복</span>
            ${roleNames.length ? `<span class="sep">·</span><span>${roleNames.map(escapeHtml).join(', ')}</span>` : ''}
          </div>
          <div class="tl-actions">
            <button class="act-toggle" onclick="toggleSchedule('${s.id}')">${enabled ? '일시중지' : '재개'}</button>
            <button class="act-delete" onclick="deleteSchedule('${s.id}')">삭제</button>
          </div>
        </div>
      </div>`;
    }).join('') + `</div>`;
  }

  window.toggleSchedule = async id => { await api(`/api/schedules/${id}/toggle`, { method: "POST" }); await loadSchedules(); };
  window.deleteSchedule = async id => {
    showModal('예약 삭제', '이 예약을 정말로 삭제하시겠습니까?', async () => {
      await api(`/api/schedules/${id}`, { method: "DELETE" });
      await loadSchedules();
    }, true);
  };

  function escapeHtml(value) {
    return String(value).replaceAll('&','&amp;').replaceAll('<','&lt;').replaceAll('>','&gt;').replaceAll('"','&quot;').replaceAll("'",'&#039;');
  }

  init();
</script>
